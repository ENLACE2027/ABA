async function guardarDatos(event) { event.preventDefault(); const checkLopd = document.getElementById('check-lopd'); if (!checkLopd.checked) { Swal.fire('Requisito Legal', 'Para enviar el formulario debes aceptar la cláusula de protección de datos.', 'warning'); return; } const requiereZonaAdaptadaGrupo = document.getElementById('check-discapacidad').checked; const tarjetas = document.querySelectorAll('.tarjeta-dinamica'); const listaParaInsertar = []; const listaParaActualizar = []; const idsParaBorrar = []; let indexVisual = 0; for (let card of tarjetas) { const idReg = card.dataset.idRegistro; if (card.classList.contains('para-borrar-registro')) { if(idReg && idReg !== "") { idsParaBorrar.push(parseInt(idReg, 10)); } continue; } const tratamiento = card.querySelector('.inv-tratamiento').value; const nombre = card.querySelector('.inv-nombre').value.trim(); const apellidos = card.querySelector('.inv-apellidos').value.trim(); const dni = card.querySelector('.inv-dni').value.trim().toUpperCase(); const fechaNac = card.querySelector('.inv-fecha-nacimiento').value; const parentesco = card.querySelector('.inv-parentesco').value; const inputObs = card.querySelector('.inv-observaciones'); const observaciones = inputObs ? inputObs.value.trim() : ""; if (!tratamiento && !nombre && !apellidos && !dni && !fechaNac && !parentesco) { continue; } indexVisual++; if (!tratamiento || !nombre || !apellidos || !dni || !fechaNac || !parentesco) { Swal.fire('Atención', `Por favor, completa todos los campos obligatorios del Acompañante ${indexVisual}.`, 'warning'); return; } if (indexVisual > 2 && !observaciones) { Swal.fire('Observación requerida', `Al superar las 2 invitaciones estándar, el Acompañante ${indexVisual} debe indicar obligatoriamente en las observaciones quién le cede la invitación (Nombre y DNI).`, 'warning'); return; } const estructuraAcompanante = { titular_dni: usuarioDni, nombre: nombre, apellidos: apellidos, dni: dni, parentesco: parentesco, tratamiento: tratamiento, fecha_nacimiento: fechaNac || null, zona_discapacidad: requiereZonaAdaptadaGrupo, observaciones: observaciones }; // Si tiene ID válido, es una actualización (Update). Si no, es nuevo (Insert). if (idReg && idReg !== "" && !isNaN(idReg)) { estructuraAcompanante.id = parseInt(idReg, 10); listaParaActualizar.push(estructuraAcompanante); } else { listaParaInsertar.push(estructuraAcompanante); } } if (listaParaInsertar.length === 0 && listaParaActualizar.length === 0 && idsParaBorrar.length === 0) { Swal.fire('Formulario Vacío', 'Introduce al menos los datos de un acompañante para guardar.', 'info'); return; } Swal.fire({ title: 'Sincronizando con el servidor...', didOpen: () => Swal.showLoading(), allowOutsideClick: false }); try { // 1. Borrar eliminados if (idsParaBorrar.length > 0) { const { error: deleteError } = await _supabase .from('invitados') .delete() .in('id', idsParaBorrar); if (deleteError) throw new Error("Error al borrar: " + deleteError.message); } // 2. Insertar nuevos registros (sin enviar el campo id) if (listaParaInsertar.length > 0) { const { error: insertError } = await _supabase .from('invitados') .insert(listaParaInsertar); if (insertError) throw new Error("Error al insertar: " + insertError.message); } // 3. Actualizar registros existentes for (let reg of listaParaActualizar) { const { error: updateError } = await _supabase .from('invitados') .update(reg) .eq('id', reg.id); if (updateError) throw new Error("Error al actualizar: " + updateError.message); } Swal.fire({ title: '¡Guardado con éxito!', text: 'La lista de acompañantes se ha actualizado de forma correcta en la base de datos.', icon: 'success', confirmButtonColor: '#4f46e5' }).then(async () => { document.getElementById('lista-invitados').innerHTML = ""; await cargarInvitadosExistentes(); }); } catch (e) { console.error("Detalle del error de sincronización:", e); Swal.fire('Error de sincronización', e.message || 'No se ha podido procesar la solicitud en Supabase.', 'error'); } }
